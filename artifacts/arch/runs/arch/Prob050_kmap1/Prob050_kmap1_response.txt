/// Top-level combinational implementation for the Prob050_kmap1 K-map.
///
/// Inputs and output are single-bit logical signals; out is high for every
/// minterm except 000.
module TopModule
  port a: in Bool;
  port b: in Bool;
  port c: in Bool;
  port out: out Bool;

  let out = a or b or c;
end module TopModule
