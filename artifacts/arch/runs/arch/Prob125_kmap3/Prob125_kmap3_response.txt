/// Combinational TopModule for the requested one-bit K-map.
///
/// Implements out = a or (c and not b); input d is ignored because it is only relevant to don't-care grouping.
module TopModule
  port a: in Bool;
  port b: in Bool;
  port c: in Bool;
  port d: in Bool;
  port out: out Bool;

  let use_c_when_b_clear: Bool = c and not b;
  let out = a or use_c_when_b_clear;
end module TopModule
