/// Top-level combinational population count module.
///
/// Counts every asserted bit in the 255-bit input and drives the exact 8-bit result in the same cycle.
module TopModule
  port in: in Vec<Bool, 255>;
  port out: out UInt<8>;

  let out = in.count(item);
end module TopModule
