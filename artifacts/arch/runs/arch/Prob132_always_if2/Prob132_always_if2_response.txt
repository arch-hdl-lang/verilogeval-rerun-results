/// Corrected combinational computer shutdown and driving-control logic.
///
/// shut_off_computer is asserted when cpu_overheated is true. keep_driving is
/// asserted only while the destination has not been reached and the gas tank is
/// not empty.
module TopModule
  port cpu_overheated: in Bool;
  port shut_off_computer: out Bool;
  port arrived: in Bool;
  port gas_tank_empty: in Bool;
  port keep_driving: out Bool;

  let shut_off_computer = cpu_overheated;
  let keep_driving = (not arrived) and (not gas_tank_empty);
end module TopModule
