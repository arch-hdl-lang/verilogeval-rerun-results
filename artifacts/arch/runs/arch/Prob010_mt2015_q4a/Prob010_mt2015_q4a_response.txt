/// One-bit combinational Boolean function for Prob010_mt2015_q4a.
///
/// Drives z as (x ^ y) & x with no clocked state.
module TopModule
  port x: in Bool;
  port y: in Bool;
  port z: out Bool;

  let z = (x ^ y) & x;
end module TopModule
