[BEGIN]
module TopModule (
    input  logic clk,
    input  logic reset,
    input  logic data,
    output logic start_shifting
);

    typedef enum logic [1:0] {
        S0,
        S1,
        S2,
        S3
    } state_t;

    state_t state;

    always @(posedge clk) begin
        if (reset) begin
            state <= S0;
            start_shifting <= 1'b0;
        end else if (start_shifting) begin
            start_shifting <= 1'b1;
        end else begin
            case (state)
                S0: begin
                    state <= data ? S1 : S0;
                end
                S1: begin
                    state <= data ? S2 : S0;
                end
                S2: begin
                    state <= data ? S2 : S3;
                end
                S3: begin
                    if (data) begin
                        start_shifting <= 1'b1;
                        state <= S1;
                    end else begin
                        state <= S0;
                    end
                end
                default: begin
                    state <= S0;
                end
            endcase
        end
    end

endmodule
[DONE]
