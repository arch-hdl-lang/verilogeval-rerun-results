[BEGIN]
module TopModule (
    input  logic        clk,
    input  logic        reset,
    output logic [31:0] q
);

    always @(posedge clk) begin
        if (reset) begin
            q <= 32'h00000001;
        end else begin
            q <= (q >> 1) ^ ({32{q[0]}} & 32'h80200003);
        end
    end

endmodule
[DONE]
