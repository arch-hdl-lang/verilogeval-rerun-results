[BEGIN]
module TopModule (
    input  logic clk,
    input  logic reset,
    input  logic in,
    output logic out
);

    logic state;

    always @(posedge clk) begin
        if (reset) begin
            state <= 1'b1;
        end else if (!in) begin
            state <= ~state;
        end else begin
            state <= state;
        end
    end

    assign out = state;

endmodule
[DONE]
