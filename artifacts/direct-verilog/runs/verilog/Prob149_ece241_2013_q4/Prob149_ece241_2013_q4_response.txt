module TopModule (
    input  logic       clk,
    input  logic       reset,
    input  logic [2:0] s,
    output logic       fr2,
    output logic       fr1,
    output logic       fr0,
    output logic       dfr
);

    logic [1:0] level;
    logic [1:0] prev_level;

    always @(*) begin
        case (s)
            3'b000: level = 2'd0;
            3'b001: level = 2'd1;
            3'b011: level = 2'd2;
            3'b111: level = 2'd3;
            default: level = 2'd0;
        endcase
    end

    always @(posedge clk) begin
        if (reset) begin
            prev_level <= 2'd0;
        end else if (level != prev_level) begin
            prev_level <= level;
        end
    end

    always @(*) begin
        fr2 = 1'b0;
        fr1 = 1'b0;
        fr0 = 1'b0;

        case (level)
            2'd0: begin
                fr2 = 1'b1;
                fr1 = 1'b1;
                fr0 = 1'b1;
            end
            2'd1: begin
                fr1 = 1'b1;
                fr0 = 1'b1;
            end
            2'd2: begin
                fr0 = 1'b1;
            end
            default: begin
                fr2 = 1'b0;
                fr1 = 1'b0;
                fr0 = 1'b0;
            end
        endcase

        dfr = (prev_level < level);
    end

endmodule
