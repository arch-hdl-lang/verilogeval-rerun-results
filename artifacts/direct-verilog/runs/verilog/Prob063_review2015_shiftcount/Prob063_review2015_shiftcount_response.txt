[BEGIN]
module TopModule (
    input  logic       clk,
    input  logic       shift_ena,
    input  logic       count_ena,
    input  logic       data,
    output logic [3:0] q
);

    always @(posedge clk) begin
        if (shift_ena) begin
            q <= {q[2:0], data};
        end else if (count_ena) begin
            q <= q - 4'd1;
        end
    end

endmodule
[DONE]
